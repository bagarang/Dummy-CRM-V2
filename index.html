// Fungsi khusus untuk menarik data Toko ke Form Aktifitas window.updateAktifitasData = function(sel) { if(!sel || sel.selectedIndex < 0) return; let opt = sel.options[sel.selectedIndex]; // Reset form jika yang dipilih adalah opsi kosong if (!opt.value) { document.getElementById('actKontak').value = ''; document.getElementById('actNama').value = ''; document.getElementById('actOrderKe').value = ''; document.getElementById('actJenis').value = ''; return; } // Isi hidden input untuk ID dan Nama document.getElementById('actKontak').value = opt.dataset.kontak || ''; document.getElementById('actNama').value = opt.dataset.nama || ''; // Cari data toko di db.toko dan auto-fill Order Ke & Jenis Kunjungan let tData = db.toko.find(t => t.kodeKontak === opt.dataset.kontak); if(tData) { document.getElementById('actOrderKe').value = tData.orderKe || ''; document.getElementById('actJenis').value = tData.jenisKunjungan || ''; } };
